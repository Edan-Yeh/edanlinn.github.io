Example：在 src/main/java/dev/edan/bankingai/LoanController.java 新增：

package dev.edan.bankingai;

import java.util.Map;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/loans")
public class LoanController {
    @GetMapping("/{id}")
    public ResponseEntity<Map<String, String>> findById(@PathVariable String id) {
        if (!"L001".equals(id)) {
            return ResponseEntity.notFound().build();
        }
        return ResponseEntity.ok(Map.of("id", "L001", "status", "ACTIVE"));
    }
}

合成資料 L001，不包含任何客戶個資或真實銀行資料。

Try：在 src/test/java/dev/edan/bankingai/LoanControllerHttpTest.java 新增：

package dev.edan.bankingai;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.web.servlet.MockMvc;
import org.springframework.test.web.servlet.setup.MockMvcBuilders;
import org.springframework.web.context.WebApplicationContext;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;

@SpringBootTest
class LoanControllerHttpTest {
    @Autowired WebApplicationContext context;

    @Test
    void existingLoanShouldReturn200() throws Exception {
        MockMvc mvc = MockMvcBuilders.webAppContextSetup(context).build();
        mvc.perform(get("/api/loans/L001"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.id").value("L001"))
            .andExpect(jsonPath("$.status").value("ACTIVE"));
    }

    @Test
    void missingLoanShouldReturn404() throws Exception {
        MockMvc mvc = MockMvcBuilders.webAppContextSetup(context).build();
        mvc.perform(get("/api/loans/UNKNOWN"))
            .andExpect(status().isNotFound());
    }
}
